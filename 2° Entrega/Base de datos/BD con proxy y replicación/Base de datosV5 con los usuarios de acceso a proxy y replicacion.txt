CREATE TABLE clientes (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL,
  origen ENUM('local','brasil','argentina') NOT NULL,
  direccion VARCHAR(255) NULL,
  usuario VARCHAR(255) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL COMMENT 'Solo hash',
  estado ENUM('activo','inactivo') NOT NULL DEFAULT 'activo',
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL,
  INDEX idx_clientes_nombre (nombre),
  INDEX idx_clientes_origen (origen)
);

CREATE TABLE usuarios_internos (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL,
  usuario VARCHAR(255) NOT NULL UNIQUE,
  password VARCHAR(255) NOT NULL COMMENT 'Solo hash',
  rol ENUM('administrativo','operativo','vendedor') NOT NULL,
  estado ENUM('activo','inactivo') NOT NULL DEFAULT 'activo',
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL
);

CREATE TABLE transportistas (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  nombre VARCHAR(255) NOT NULL,
  email VARCHAR(255) NULL,
  contacto VARCHAR(255) NULL,
  estado ENUM('activo','inactivo') NOT NULL DEFAULT 'activo',
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL,
  INDEX idx_transportistas_nombre (nombre),
  INDEX idx_transportistas_email (email)
);

CREATE TABLE fichas (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  num_ficha VARCHAR(50) NULL UNIQUE COMMENT 'Interno; no se muestra al cliente',
  tipo_operacion ENUM('importacion','exportacion') NOT NULL,
  cliente_id BIGINT UNSIGNED NOT NULL,
  transportista_id BIGINT UNSIGNED NULL,
  vendedor_id BIGINT UNSIGNED NULL,
  estado ENUM('iniciada','en_curso','en_frontera','entregada','anulada') NOT NULL DEFAULT 'iniciada',
  num_crt VARCHAR(255) NULL UNIQUE,
  -- Datos del cliente en la operación
  cliente_nombre VARCHAR(255) NULL,
  cliente_contacto VARCHAR(255) NULL,
  cliente_telefono VARCHAR(50) NULL,
  cliente_ruc_rut VARCHAR(50) NULL,
  -- Exportador y mercadería
  nombre_exportador VARCHAR(255) NULL,
  contacto_exportador_nombre VARCHAR(255) NULL,
  contacto_exportador_email VARCHAR(255) NULL,
  telefono_exportador VARCHAR(50) NULL,
  despachante_exportador VARCHAR(255) NULL,
  mercaderia VARCHAR(255) NULL,
  tipo_carga ENUM('consolidada','completa') NULL,
  peso_bruto DECIMAL(10,2) NULL COMMENT 'kg',
  volumen DECIMAL(10,2) NULL COMMENT 'm3',
  tipo_bulto VARCHAR(255) NULL,
  cantidad_bultos INT UNSIGNED NULL,
  num_factura VARCHAR(255) NULL,
  -- Carga y descarga
  ciudad_carga VARCHAR(255) NULL,
  frontera VARCHAR(255) NULL,
  coleta VARCHAR(255) NULL,
  ciudad_descarga VARCHAR(255) NULL,
  direccion_descarga VARCHAR(255) NULL,
  telefono_descarga VARCHAR(50) NULL,
  contacto_descarga VARCHAR(255) NULL,
  email_descarga VARCHAR(255) NULL,
  -- Despachante local
  despachante_local_nombre VARCHAR(255) NULL,
  despachante_local_email VARCHAR(255) NULL,
  despachante_local_telefono VARCHAR(50) NULL,
  despachante_local_direccion VARCHAR(255) NULL,
  despachante_local_gestor VARCHAR(255) NULL,
  despachante_local_gestor_telefono VARCHAR(50) NULL,
  despachante_local_contacto VARCHAR(255) NULL,
  dua VARCHAR(6) NULL COMMENT 'Máx. 6 dígitos',
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL,
  FOREIGN KEY (cliente_id) REFERENCES clientes(id),
  FOREIGN KEY (transportista_id) REFERENCES transportistas(id),
  FOREIGN KEY (vendedor_id) REFERENCES usuarios_internos(id),
  CONSTRAINT chk_fichas_dua CHECK (dua IS NULL OR dua REGEXP '^[0-9]{1,6}$'),
  INDEX idx_fichas_dua (dua),
  INDEX idx_fichas_estado (estado),
  INDEX idx_fichas_created (created_at)
);

CREATE TABLE seguimientos (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  ficha_id BIGINT UNSIGNED NOT NULL,
  etapa ENUM('carga','deposito','frontera','liberacion','entrega') NULL,
  fecha DATE NULL,
  canal_aduanero ENUM('verde','naranja','rojo') NULL,
  observacion TEXT NULL,
  notificar_cliente BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL,
  FOREIGN KEY (ficha_id) REFERENCES fichas(id) ON DELETE RESTRICT,
  CONSTRAINT chk_seguimientos_canal CHECK (canal_aduanero IS NULL OR etapa = 'frontera'),
  CONSTRAINT chk_seguimiento_con_datos CHECK (etapa IS NOT NULL OR fecha IS NOT NULL OR observacion IS NOT NULL),
  CONSTRAINT chk_seguimiento_etapa_fecha CHECK ((etapa IS NULL AND fecha IS NULL) OR (etapa IS NOT NULL AND fecha IS NOT NULL)),
  INDEX idx_seguimientos_ficha_fecha (ficha_id, fecha)
);

-- No está en la especificación; se conserva por si la web la usa.
CREATE TABLE notificaciones (
  id BIGINT UNSIGNED AUTO_INCREMENT PRIMARY KEY,
  ficha_id BIGINT UNSIGNED NOT NULL,
  tipo VARCHAR(255) NOT NULL,
  mensaje TEXT NOT NULL,
  leida BOOLEAN NOT NULL DEFAULT FALSE,
  created_at TIMESTAMP NULL,
  updated_at TIMESTAMP NULL,
  FOREIGN KEY (ficha_id) REFERENCES fichas(id) ON DELETE RESTRICT
);

-- Usuario único entre clientes y usuarios internos
DELIMITER $$
CREATE TRIGGER trg_clientes_usuario_bi BEFORE INSERT ON clientes
FOR EACH ROW BEGIN
  IF EXISTS (SELECT 1 FROM usuarios_internos WHERE usuario = NEW.usuario) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El usuario ya existe como usuario interno';
  END IF;
END$$
CREATE TRIGGER trg_clientes_usuario_bu BEFORE UPDATE ON clientes
FOR EACH ROW BEGIN
  IF NEW.usuario <> OLD.usuario
     AND EXISTS (SELECT 1 FROM usuarios_internos WHERE usuario = NEW.usuario) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El usuario ya existe como usuario interno';
  END IF;
END$$
CREATE TRIGGER trg_internos_usuario_bi BEFORE INSERT ON usuarios_internos
FOR EACH ROW BEGIN
  IF EXISTS (SELECT 1 FROM clientes WHERE usuario = NEW.usuario) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El usuario ya existe como cliente';
  END IF;
END$$
CREATE TRIGGER trg_internos_usuario_bu BEFORE UPDATE ON usuarios_internos
FOR EACH ROW BEGIN
  IF NEW.usuario <> OLD.usuario
     AND EXISTS (SELECT 1 FROM clientes WHERE usuario = NEW.usuario) THEN
    SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = 'El usuario ya existe como cliente';
  END IF;
END$$
DELIMITER ;
